//! Minimal reference executor for the G2 oracle-feasibility probe
//! (issue #304). This is NOT an H0–H4 candidate implementation and NOT a
//! benchmark: it is the smallest honest witness of the frozen lifecycle
//! contract (Offered→Admitted→Running→Terminal) plus the three separated
//! actions request_cancel / await_settlement / await_quiescence, written
//! with plain native threads + Mutex/Condvar (the repository's current
//! mechanism vocabulary — zero new dependencies).
//!
//! Determinism discipline (concurrency-test-design):
//!   - Every ordering the oracle judges is expressed in ledger sequence
//!     numbers (the one serialization boundary), never wall-clock.
//!   - The cancel milestone is admission-count based (workload-owned),
//!     independent of executor speed.
//!   - The synthetic I/O wait (`std::thread::sleep`) is a WORKLOAD
//!     parameter (simulated pacing), never a synchronization premise.
//!   - M5's parked-worker flag is observed through a real Condvar edge;
//!     the 30 s park timeout is a cleanup watchdog, never evidence.

use crate::ledger::{EffectKind, Ledger, TerminalKind};
use crate::oracle::fold_meta;
use crate::workload::{self, BatchSpec};
use std::collections::{HashSet, VecDeque};
use std::sync::{Arc, Condvar, Mutex};
use std::thread::JoinHandle;
use std::time::Duration;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Mutant {
    None,
    /// M1: an admitted item's obligation is silently lost (queue skip +
    /// bookkeeping forget) — admitted but never terminal.
    M1DropAdmitted,
    /// M2: duplicate ResultPublished effect for one item.
    M2DuplicateEffect,
    /// M3: TransformDone effect published AFTER the item's terminal ACK.
    M3LateEffect,
    /// M4: worker publishes Abandoned with no cancel ever requested for
    /// the item's group (premature/forged abandon).
    M4PrematureAbandon,
    /// M5: coordinator returns quiescence while one worker is still
    /// parked (never joined, exit never recorded).
    M5QuiesceWithoutJoin,
    /// M6: content corruption — one item transformed with cpu_blocks-1
    /// (real work reduced); digests change and must be caught by O3.
    M6CorruptContent,
    /// M7: post-cancel admission — a group-7 item admitted AFTER the
    /// group cancel record (must be caught by O7).
    M7PostCancelAdmission,
    /// M8: wait skipped — one IoWait item's synthetic wait and its
    /// IoWaitDone effect both omitted (must be caught by O2).
    M8SkipWait,
    /// M9: error-lane payload forgery — a fault item publishes a
    /// fabricated contribution instead of its REF value (must be caught
    /// by O3's Err-branch pinning).
    M9ForgeErrPayload,
}

pub struct ProbeConfig {
    /// P — execution parallelism budget (worker threads).
    pub workers: usize,
    /// B — admitted-but-not-terminal bound (backpressure).
    pub bound: usize,
    /// k — workload-owned causal milestone: request_cancel(group 7)
    /// fires immediately after the k-th ItemAdmitted.
    pub cancel_after_admissions: u32,
    pub mutant: Mutant,
}

impl Default for ProbeConfig {
    fn default() -> Self {
        ProbeConfig { workers: 4, bound: 8, cancel_after_admissions: 24, mutant: Mutant::None }
    }
}

const CANCEL_GROUP: u32 = 7;
/// Fixed mutant target items. ALL targets are in group 1 (never
/// cancelled by the milestone) and admitted before the k-th admission,
/// so each mutation is reachable in EVERY schedule — no mutant can be
/// swallowed by the legal abandon path.
const M1_ITEM: u32 = 20;
const M2_ITEM: u32 = 10;
const M3_ITEM: u32 = 16;
const M4_ITEM: u32 = 22;
const M5_WORKER: u64 = 4; // worker id that parks (1-based; P = 4)
// Round-1 review mutants: group-1 / reachable-by-construction targets.
// M6_ITEM: CpuHeavy, no fault, group 1. M7_ITEM: group 7, admitted after
// the milestone. M8_ITEM: IoWait class (wait_us=2000), group 1.
const M6_ITEM: u32 = 26;
const M7_ITEM: u32 = 27;
const M8_ITEM: u32 = 25;
// M9_ITEM: injected-fault item (item_id%16==13), group 1.
const M9_ITEM: u32 = 13;

struct ExecState {
    queue: VecDeque<u32>,
    /// B accounting: admitted minus terminal (the real counted set).
    admitted_not_terminal: usize,
    /// Obligation bookkeeping: admitted items not yet terminal.
    /// (M1's defect is to drop an item from here AND from the queue.)
    pending: HashSet<u32>,
    cancelled_groups: HashSet<u32>,
    producer_done: bool,
    /// Fan-in sink: (item_id, contribution) as actually received.
    results: Vec<(u32, u64)>,
    /// Worker ids that recorded their exit (drain complete).
    exited: HashSet<u64>,
    /// M5 probe: id of the worker parked at drain, set under this lock.
    parked_worker: Option<u64>,
    release_parked: bool,
}

pub struct ReferenceExecutor {
    cfg: ProbeConfig,
    spec: BatchSpec,
    ledger: Arc<Ledger>,
    state: Arc<(Mutex<ExecState>, Condvar, Condvar, Condvar, Condvar)>,
    // (work_cv, slot_cv, settle_cv, park_cv) — one lock domain.
    handles: Vec<JoinHandle<()>>,
    worker_ids: Vec<u64>,
}

impl ReferenceExecutor {
    pub fn new(cfg: ProbeConfig, spec: BatchSpec, ledger: Arc<Ledger>) -> Self {
        ReferenceExecutor {
            cfg,
            spec,
            ledger,
            state: Arc::new((
                Mutex::new(ExecState {
                    queue: VecDeque::new(),
                    admitted_not_terminal: 0,
                    pending: HashSet::new(),
                    cancelled_groups: HashSet::new(),
                    producer_done: false,
                    results: Vec::new(),
                    exited: HashSet::new(),
                    parked_worker: None,
                    release_parked: false,
                }),
                Condvar::new(),
                Condvar::new(),
                Condvar::new(),
                Condvar::new(),
            )),
            handles: Vec::new(),
            worker_ids: Vec::new(),
        }
    }

    /// Spawn P workers (each records ExecutorSpawned at the true spawn
    /// boundary), then run the admission phase (producer role) on the
    /// calling thread, including the workload-owned cancel milestone.
    pub fn start(&mut self) {
        for w in 1..=self.cfg.workers as u64 {
            let state = Arc::clone(&self.state);
            let ledger = Arc::clone(&self.ledger);
            let spec = self.spec.clone();
            let mutant = self.cfg.mutant;
            let handle = std::thread::Builder::new()
                .name(format!("probe-worker-{w}"))
                .spawn(move || worker_main(w, mutant, spec, ledger, state))
                .expect("spawn worker");
            self.ledger.effect(self.spec.batch_id, None, EffectKind::ExecutorSpawned, w, 0);
            self.handles.push(handle);
            self.worker_ids.push(w);
        }
        self.admission_phase();
        let (lock, work_cv, _slot, settle_cv, _park) = &*self.state;
        {
            let mut st = lock.lock().unwrap();
            st.producer_done = true;
            drop(st);
            work_cv.notify_all();
            // settle waiters key off (producer_done && pending.is_empty):
            // the flag flip alone can make the predicate true, so they
            // must be woken here too (round-1 review lost-wakeup fix).
            settle_cv.notify_all();
        }
    }

    /// Producer role: admit items under the B bound; the k-th admission
    /// triggers request_cancel(CANCEL_GROUP) — the workload-owned causal
    /// milestone (admission-count based, executor-speed independent).
    fn admission_phase(&mut self) {
        let (lock, _work, slot_cv, _settle, _park) = &*self.state;
        let mut admitted = 0u32;
        for item in &self.spec.items {
            let id = item.item_id;
            let group = item.cancellation_group;
            // Backpressure: wait until admitted-but-not-terminal < B.
            // (Real counted bound — not queue capacity alone.)
            let mut st = loop {
                let guard = lock.lock().unwrap();
                if guard.admitted_not_terminal < self.cfg.bound {
                    break guard;
                }
                let (guard, _t) = slot_cv.wait_timeout(guard, Duration::from_millis(50)).unwrap();
                drop(guard);
            };
            if st.cancelled_groups.contains(&group)
                && !(self.cfg.mutant == Mutant::M7PostCancelAdmission && id == M7_ITEM)
            {
                continue; // admission cut: stays Offered, no obligation
            }
            self.ledger
                .effect(self.spec.batch_id, Some(id), EffectKind::ItemAdmitted, group as u64, 0);
            admitted += 1;
            if self.cfg.mutant == Mutant::M1DropAdmitted && id == M1_ITEM {
                // M1: the obligation is LOST after admission — never
                // queued, never tracked, never terminal.
            } else {
                st.queue.push_back(id);
                st.pending.insert(id);
                st.admitted_not_terminal += 1;
            }
            drop(st);
            if admitted == self.cfg.cancel_after_admissions {
                self.request_cancel(CANCEL_GROUP);
            }
        }
    }

    /// request_cancel(): records intent + cuts admission for the group.
    /// Returns having recorded the intent — promises NOTHING about any
    /// in-flight item (they complete or are abandoned by their owner).
    pub fn request_cancel(&self, group: u32) {
        let (lock, work_cv, _s, _se, _p) = &*self.state;
        let mut st = lock.lock().unwrap();
        if st.cancelled_groups.insert(group) {
            // Cancel record appended while holding the state lock: any
            // worker can observe the flag only after this append, so every
            // Abandoned ACK for the group lands at a later sequence.
            self.ledger
                .effect(self.spec.batch_id, None, EffectKind::CancelRequested, group as u64, 0);
        }
        drop(st);
        work_cv.notify_all();
    }

    /// await_settlement(): returns when every ADMITTED item has exactly
    /// one terminal ACK. Promises nothing about executors still running.
    pub fn await_settlement(&self) {
        let (lock, _w, _s, settle_cv, _p) = &*self.state;
        let mut st = lock.lock().unwrap();
        while !(st.producer_done && st.pending.is_empty()) {
            st = settle_cv.wait(st).unwrap();
        }
    }

    /// await_quiescence(): joins every executor (M5 mutant: skips the
    /// parked one), folds the fan-in sink, publishes the aggregate and
    /// the quiescence receipt. The receipt is the LAST ledger record.
    pub fn await_quiescence(&mut self) {
        let (lock, work_cv, _s, _se, park_cv) = &*self.state;
        if self.cfg.mutant == Mutant::M5QuiesceWithoutJoin {
            // Wait until the probe worker is observably parked AND every
            // other worker has recorded its exit (real Condvar edges, not
            // sleeps): then "return quiescence" without joining worker 4.
            // Only worker 4 is unjoined at the receipt — the RED is then
            // deterministic and localized to O6.
            let mut st = lock.lock().unwrap();
            while !(st.parked_worker == Some(M5_WORKER)
                && st.exited.len() == self.cfg.workers - 1)
            {
                st = work_cv.wait(st).unwrap();
            }
            drop(st);
        } else {
            for h in self.handles.drain(..) {
                let _ = h.join();
            }
        }
        // Fan-in: fold actually-received contributions in item_id order.
        let pairs = {
            let mut st = lock.lock().unwrap();
            st.results.sort_by(|a, b| a.0.cmp(&b.0));
            st.results.clone()
        };
        let agg = workload::aggregate(&pairs);
        self.ledger
            .effect(self.spec.batch_id, None, EffectKind::AggregatePublished, agg, 0);
        self.ledger
            .effect(self.spec.batch_id, None, EffectKind::QuiescenceReceipt, 0, 0);
        if self.cfg.mutant == Mutant::M5QuiesceWithoutJoin {
            // Cleanup only (after the harness has taken its oracle
            // snapshot): release and join the parked worker so the test
            // process leaks nothing. The 30 s watchdog in worker_main is
            // pure cleanup safety.
            let mut st = lock.lock().unwrap();
            st.release_parked = true;
            drop(st);
            park_cv.notify_all();
            for h in self.handles.drain(..) {
                let _ = h.join();
            }
        }
    }
}

fn worker_main(
    worker_id: u64,
    mutant: Mutant,
    spec: BatchSpec,
    ledger: Arc<Ledger>,
    state: Arc<(Mutex<ExecState>, Condvar, Condvar, Condvar, Condvar)>,
) {
    let (lock, work_cv, _slot, _settle_cv, park_cv) = &*state;
    loop {
        let mut st = lock.lock().unwrap();
        while st.queue.is_empty() && !st.producer_done {
            st = work_cv.wait(st).unwrap();
        }
        if st.queue.is_empty() && st.producer_done {
            // Drain complete — the only legal exit funnel.
            if mutant == Mutant::M5QuiesceWithoutJoin && worker_id == M5_WORKER {
                st.parked_worker = Some(worker_id);
                drop(st);
                work_cv.notify_all();
                let mut g = lock.lock().unwrap();
                let deadline = std::time::Instant::now() + Duration::from_secs(30);
                while !g.release_parked {
                    let (ng, tmo) =
                        park_cv.wait_timeout(g, deadline.saturating_duration_since(std::time::Instant::now())).unwrap();
                    g = ng;
                    if tmo.timed_out() {
                        break; // cleanup watchdog — never a correctness premise
                    }
                }
            } else {
                drop(st);
            }
            let mut g = lock.lock().unwrap();
            g.exited.insert(worker_id);
            ledger.effect(spec.batch_id, None, EffectKind::ExecutorExited, worker_id, worker_id);
            drop(g);
            work_cv.notify_all();
            return;
        }
        let item_id = st.queue.pop_front().expect("non-empty queue");
        let cancelled = {
            let group = spec.items[item_id as usize].cancellation_group;
            st.cancelled_groups.contains(&group)
        };
        drop(st);

        // Abandon branch: LEGAL when the item's group cancel was already
        // recorded (worker owns the item at dequeue); FORGED in the M4
        // arm (group 1 is never cancelled — O5 must fire).
        if cancelled || (mutant == Mutant::M4PrematureAbandon && item_id == M4_ITEM) {
            finish_terminal(&spec, &ledger, &state, item_id, TerminalKind::Abandoned, worker_id, None);
            continue;
        }
        process_item(worker_id, mutant, &spec, &ledger, &state, item_id);
    }
}

fn process_item(
    worker_id: u64,
    mutant: Mutant,
    spec: &BatchSpec,
    ledger: &Ledger,
    state: &Arc<(Mutex<ExecState>, Condvar, Condvar, Condvar, Condvar)>,
    item_id: u32,
) {
    let s = &spec.items[item_id as usize];
    // M8: the synthetic wait and its effect are BOTH omitted — the
    // wait dimension must be adjudicated (round-1 review).
    let skip_wait = mutant == Mutant::M8SkipWait && item_id == M8_ITEM;
    // Synthetic logical I/O wait — a WORKLOAD parameter (simulated
    // pacing). Blocking by design for this native-thread witness.
    if s.workload_class.wait_us() > 0 && !skip_wait {
        std::thread::sleep(Duration::from_micros(s.workload_class.wait_us()));
        ledger.effect(spec.batch_id, Some(item_id), EffectKind::IoWaitDone, s.workload_class.wait_us(), worker_id);
    }
    let bytes = workload::generate_input(spec.seed, item_id);
    // M6: one block of CPU work quietly removed — digests must shift.
    let cpu_blocks = if mutant == Mutant::M6CorruptContent && item_id == M6_ITEM {
        s.workload_class.cpu_blocks() - 1
    } else {
        s.workload_class.cpu_blocks()
    };
    let (meta, peak) = workload::transform(item_id, &bytes, cpu_blocks);
    let content_digest = workload::content_digest(item_id, &bytes);
    let mut contribution = workload::contribution(&content_digest, &meta, peak);
    if mutant == Mutant::M9ForgeErrPayload && item_id == M9_ITEM {
        // M9: the error lane carries a fabricated contribution.
        contribution = 0x00DE_ADB_EF00_DEAD;
    }
    let meta_fold = fold_meta(&meta);

    match mutant {
        Mutant::M3LateEffect if item_id == M3_ITEM => {
            // M3: publish result + terminal ACK, THEN a TransformDone
            // effect — an effect after the item's own terminal.
            ledger.effect(spec.batch_id, Some(item_id), EffectKind::ResultPublished, contribution, worker_id);
            let term = match s.fault {
                Some(c) => TerminalKind::CompletedErr(c),
                None => TerminalKind::CompletedOk,
            };
            finish_terminal(spec, ledger, state, item_id, term, worker_id, Some(contribution));
            ledger.effect(spec.batch_id, Some(item_id), EffectKind::TransformDone, meta_fold, worker_id);
            return;
        }
        Mutant::M2DuplicateEffect if item_id == M2_ITEM => {
            ledger.effect(spec.batch_id, Some(item_id), EffectKind::TransformDone, meta_fold, worker_id);
            ledger.effect(spec.batch_id, Some(item_id), EffectKind::ResultPublished, contribution, worker_id);
            // M2: the same protected effect published a second time.
            ledger.effect(spec.batch_id, Some(item_id), EffectKind::ResultPublished, contribution, worker_id);
            let term = match s.fault {
                Some(c) => TerminalKind::CompletedErr(c),
                None => TerminalKind::CompletedOk,
            };
            finish_terminal(spec, ledger, state, item_id, term, worker_id, Some(contribution));
            return;
        }
        _ => {}
    }

    ledger.effect(spec.batch_id, Some(item_id), EffectKind::TransformDone, meta_fold, worker_id);
    ledger.effect(spec.batch_id, Some(item_id), EffectKind::ResultPublished, contribution, worker_id);
    let term = match s.fault {
        Some(c) => TerminalKind::CompletedErr(c),
        None => TerminalKind::CompletedOk,
    };
    finish_terminal(spec, ledger, state, item_id, term, worker_id, Some(contribution));
}

/// Append the terminal ACK (worker = item terminal owner for Completed*;
/// for Abandoned the owner is whoever holds the item when the cancel is
/// observed), update the shared obligation/sink state, wake the
/// settlement/backpressure waiters.
fn finish_terminal(
    spec: &BatchSpec,
    ledger: &Ledger,
    state: &Arc<(Mutex<ExecState>, Condvar, Condvar, Condvar, Condvar)>,
    item_id: u32,
    term: TerminalKind,
    owner: u64,
    contribution: Option<u64>,
) {
    let digest = contribution.unwrap_or(0);
    ledger.terminal_ack(spec.batch_id, item_id, term, digest, owner);
    let (lock, _work_cv, slot_cv, settle_cv, _park) = &**state;
    let mut st = lock.lock().unwrap();
    if st.pending.remove(&item_id) {
        st.admitted_not_terminal = st.admitted_not_terminal.saturating_sub(1);
    }
    if let Some(c) = contribution {
        st.results.push((item_id, c));
    }
    drop(st);
    slot_cv.notify_all();
    settle_cv.notify_all();
}

```rust
//! G2 oracle-feasibility probe harness (issue #304, MVCS-0 R2-0).
//!
//! Arms:
//!   GREEN — reference executor, expect all checks green.
//!   M1..M5 — one bounded fault variant each, expect the SAME oracle to
//!            go red (each arm must be detected by its designated check).
//!   REFCHECK — REF determinism + aggregate self-consistency (no threads).
//!
//! Exit code 0 = expectation met (green arm green / mutant arm red);
//! exit code 1 = expectation NOT met (oracle failure or mutant escaped).
//! One batch per process: batch isolation (O8) is meaningful per run.

mod ledger;
mod oracle;
mod reference_executor;
mod workload;

use ledger::Ledger;
use oracle::OracleVerdict;
use reference_executor::{Mutant, ProbeConfig, ReferenceExecutor};
use std::sync::Arc;
use workload::BatchSpec;

fn parse_mutant(arg: &str) -> Option<(Mutant, &'static str, u32)> {
    match arg {
        "GREEN" => Some((Mutant::None, "reference (expect GREEN)", 24)),
        "GREEN_NC" => {
            Some((Mutant::None, "reference, NO cancel (expect GREEN; REF-pinned full aggregate)", u32::MAX))
        }
        "M1" => Some((Mutant::M1DropAdmitted, "drop admitted item (expect RED: O1)", 24)),
        "M2" => Some((Mutant::M2DuplicateEffect, "duplicate effect (expect RED: O2)", 24)),
        "M3" => Some((Mutant::M3LateEffect, "effect after item terminal (expect RED: O4)", 24)),
        "M4" => Some((Mutant::M4PrematureAbandon, "forged abandon (expect RED: O5)", 24)),
        "M5" => Some((Mutant::M5QuiesceWithoutJoin, "quiesce without join (expect RED: O6)", 24)),
        "M6" => Some((Mutant::M6CorruptContent, "content corruption cpu_blocks-1 (expect RED: O3)", 24)),
        "M7" => Some((Mutant::M7PostCancelAdmission, "post-cancel admission (expect RED: O7)", 24)),
        "M8" => Some((Mutant::M8SkipWait, "skipped synthetic wait (expect RED: O2)", 24)),
        "M9" => Some((Mutant::M9ForgeErrPayload, "error-lane payload forgery (expect RED: O3)", 24)),
        _ => None,
    }
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    if args.len() != 2 {
        eprintln!("usage: probe <GREEN|M1|M2|M3|M4|M5|REFCHECK>");
        std::process::exit(2);
    }
    let arm = args[1].as_str();

    if arm == "REFCHECK" {
        refcheck();
        return;
    }

    let Some((mutant, desc, cancel_at)) = parse_mutant(arm) else {
        eprintln!("unknown arm: {arm}");
        std::process::exit(2);
    };

    let spec = BatchSpec::probe_batch();
    let ledger = Arc::new(Ledger::new());
    let cfg = ProbeConfig { workers: 4, bound: 8, cancel_after_admissions: cancel_at, mutant };

    let t0 = std::time::Instant::now();
    let mut ex = ReferenceExecutor::new(cfg, spec.clone(), Arc::clone(&ledger));
    ex.start(); // admission phase + workers (cancel milestone inside)
    ex.await_settlement();
    let settled = t0.elapsed();
    ex.await_quiescence();
    let quiesced = t0.elapsed();

    // M5 note: its cleanup (release+join of the parked worker) happens
    // INSIDE await_quiescence, so the snapshot below already contains
    // the parked worker's post-receipt ExecutorExited record — which is
    // exactly what O6/O10 adjudicate (exit strictly after the receipt).
    let verdict = oracle::adjudicate(&spec, &ledger);
    print_verdict(arm, desc, &verdict, settled, quiesced, ledger.last_seq(), cancel_at);

    let expect_green = arm.starts_with("GREEN");
    let green = verdict.green();
    if green == expect_green {
        println!("ARM_RESULT = MET ({} {})", arm, if green { "GREEN" } else { "RED" });
        std::process::exit(0);
    } else {
        println!("ARM_RESULT = VIOLATION ({} {})", arm, if green { "GREEN" } else { "RED" });
        std::process::exit(1);
    }
}

fn print_verdict(
    arm: &str,
    desc: &str,
    verdict: &OracleVerdict,
    settled: std::time::Duration,
    quiesced: std::time::Duration,
    last_seq: u64,
    cancel_at: u32,
) {
    println!("ARM           = {arm} ({desc})");
    println!("BATCH         = N=48, P=4, B=8, cancel@admission-{cancel_at}(group 7)");
    println!("LEDGER_SEQ    = {last_seq}");
    println!("SETTLE_T      = {settled:?}   (diagnostic only, not evidence)");
    println!("QUIESCE_T     = {quiesced:?}   (diagnostic only, not evidence)");
    for c in &verdict.checks {
        println!("  {:<36} {}  {}", c.check, if c.pass { "PASS" } else { "FAIL" }, c.detail);
    }
    let failed = verdict.failed();
    println!("FAILED_CHECKS = {}", failed.len());
    println!("VERDICT       = {}", if verdict.green() { "GREEN" } else { "RED" });
}

/// REF determinism self-check (M0-tier, no threads involved): the same
/// (seed, item) must always yield the same REF, and the aggregate must
/// be invariant under input permutation (order independence).
fn refcheck() {
    let spec = BatchSpec::probe_batch();
    let mut first: Vec<(u32, [u8; 32], u64)> = Vec::new();
    for s in &spec.items {
        let r = workload::ref_result(spec.seed, s);
        first.push((s.item_id, r.metadata_digest, r.aggregation_contribution));
    }
    let mismatch = spec
        .items
        .iter()
        .filter(|s| {
            let r = workload::ref_result(spec.seed, s);
            let f = &first[s.item_id as usize];
            f.1 != r.metadata_digest || f.2 != r.aggregation_contribution
        })
        .count();
    let contributions: Vec<(u32, u64)> = first.iter().map(|(i, _, c)| (*i, *c)).collect();
    let mut shuffled = contributions.clone();
    // deterministic rotation (not RNG) — permutation of the same pairs
    shuffled.rotate_left(7);
    let agg_a = workload::aggregate(&contributions);
    let agg_b = workload::aggregate(&shuffled);
    println!("REF_DETERMINISM   = {}", if mismatch == 0 { "STABLE" } else { "UNSTABLE" });
    println!("AGG_ORDER_INVAR   = {}", if agg_a == agg_b { "YES" } else { "NO" });
    println!("AGG_VALUE         = {agg_a:#018x}");
    let ok = mismatch == 0 && agg_a == agg_b;
    std::process::exit(if ok { 0 } else { 1 });
}
```
